// ─────────────────────────────────────────────────────────────────────────────
// Story template — COPY this file next to a new component and rename it
// `<component>.stories.tsx` (the `.txt` extension keeps this skeleton out of the
// build). Convention: every @repo/ui component ships with a co-located story so it
// shows up in `pnpm --filter @repo/ui storybook`.
//
// 1. Replace `MyComponent` / `MyComponentProps` with your component + props type.
// 2. Always import from '@repo/ui' (the public export surface), never deep paths.
// 3. Set `title` to `Elements/…`, `Layout/…`, or `Sections/…` to match the folder.
//    (`Pages/…` is reserved for composed page previews — several sections stacked on
//    static props, the artifact the Design Mode gate is asked about. Those set
//    `parameters: { layout: 'fullscreen' }`; see .claude/skills/design-mode/SKILL.md.)
// 4. Add an `argTypes` control for every union prop so it's tweakable in the panel.
// ─────────────────────────────────────────────────────────────────────────────
import type { Meta, StoryObj } from '@storybook/react-vite'
import { MyComponent, type MyComponentProps } from '@repo/ui'

const variants: NonNullable<MyComponentProps['variant']>[] = ['default']

const meta = {
  title: 'Elements/MyComponent',
  component: MyComponent,
  // 'autodocs' generates a Docs page from props + stories. Use 'fullscreen'
  // layout for full-width sections; omit it for centered elements.
  tags: ['autodocs'],
  args: {
    // default args render in the Playground story
    variant: 'default',
  },
  argTypes: {
    variant: { control: 'select', options: variants },
  },
} satisfies Meta<typeof MyComponent>

export default meta
type Story = StoryObj<typeof meta>

export const Playground: Story = {}

export const Variants: Story = {
  render: (args) => (
    <div className="flex flex-wrap items-center gap-3">
      {variants.map((variant) => (
        <MyComponent key={variant} {...args} variant={variant} />
      ))}
    </div>
  ),
}
